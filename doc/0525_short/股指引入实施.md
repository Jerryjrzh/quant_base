#### 第二步：编写多期指状态评估函数 (在 worker 内或外部工具类中)

每次对 `historical_df` (T日) 或在未来的每一天，调用这个探测器，判定当天的宏观状态。

```python
def evaluate_multi_index_risk(indices_dict, target_date):
    """
    多期指共振与背离风险评估引擎
    返回: risk_level (0-100), market_state_str, price_discount_multiplier, max_positions
    """
    drops = {}
    is_ma20_broken = {}
    
    for name, df in indices_dict.items():
        if df is None or target_date not in df.index:
            return 50, 'DATA_MISSING', 1.0, 15
            
        loc = df.index.get_loc(target_date)
        if loc < 20: continue
        
        close = df.iloc[loc]['close']
        prev_close = df.iloc[loc-1]['close']
        ma20 = df['close'].iloc[loc-20:loc].mean()
        
        drops[name] = (close - prev_close) / prev_close
        is_ma20_broken[name] = close < ma20

    # 防御缺失数据
    if len(drops) < 4: return 50, 'NORMAL', 1.0, 15

    # 1. 系统性崩盘判定 (全部暴跌)
    if all(d < -0.03 for d in drops.values()):
        return 100, 'SYSTEMIC_CRASH', 0.90, 0  # 彻底断路，拒绝买入，挂单极深
        
    # 2. 致命剪刀差判定 (拉高权重，中小盘雪崩)
    # IH/IF 跌幅极小甚至上涨，但 IC/IM 出现超过 2% 的大跌
    if (drops['IH'] > -0.005 or drops['IF'] > -0.005) and (drops['IC'] < -0.02 or drops['IM'] < -0.02):
        return 90, 'COVERT_DISTRIBUTION', 0.94, 3  # 二八分化，仓位极度缩减到3只，索要6%折扣

    # 3. 中小盘主跌浪判定 (微盘期指破位)
    if is_ma20_broken['IM'] and is_ma20_broken['IC'] and drops['IM'] < 0:
        return 70, 'SME_BEAR_TREND', 0.96, 5  # 中小盘空仓期，仓位降至5只，索要4%折扣

    # 4. 全面多头或常规震荡
    return 10, 'NORMAL', 1.0, 15 # 正常态，满仓15只，无额外折扣惩罚

```
