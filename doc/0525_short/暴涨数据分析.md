def get_stats(df, col_name):
    return df[col_name].describe(percentiles=[.05, .10, .25, .50, .75, .90, .95]).round(4)

results = []
for period in periods:
    for metric in metrics:
        col = f"{period}_{metric}"
        if col in df_up_clean.columns:

            stats_down = get_stats(df_down_clean, col)

# Let's combine both dataframes for a global threshold base, or use df_up for "bull" specific thresholds
df_all = pd.concat([df_up_clean, df_down_clean])

def get_thresholds(df, period):
    body = df[f'{period}_实体涨幅%']
    u_shadow = df[f'{period}_上影线%']
    l_shadow = df[f'{period}_下影线%']
    
    # U (Big Bull) and D (Big Bear) thresholds based on absolute percentiles
    abs_body = body.abs()
    U_90 = np.percentile(abs_body, 90)
    U_80 = np.percentile(abs_body, 80)
    
    # Shadow thresholds (Upper for T, Lower for B)
    T_90 = np.percentile(u_shadow, 90)
    B_90 = np.percentile(l_shadow, 90)
    
    print(f"--- {period} Thresholds ---")
    print(f"  Absolute Body >= 90th percentile: {U_90:.2f}% (Use for U/D)")
