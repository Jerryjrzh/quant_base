is_golden_pit = False

# 条件2: 经历了极致缩量阴跌
vol_ma20 = df['volume'].rolling(20).mean().iloc[-1]

# 条件3: 均线破位但长期均线向上 (骗线)
ma30, ma60, ma240 = current['ma30'], current['ma60'], current['ma240']
is_fake_breakdown = (current['close'] < ma30) and (current['close'] < ma60) and (current['close'] > ma240) and (ma240 > df['ma240'].iloc[-20])

# 条件4: 水下点火 (KDJ超卖 + MACD刚翻红)
